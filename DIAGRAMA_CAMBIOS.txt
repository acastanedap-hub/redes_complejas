8. PARÁMETROS DE CONFIGURACIÓN
═════════════════════════════════════════════════════════════════════════════════

En src/config.py ahora tienes:

    # ESTRICTO: 90° ± 8 grados (POR DEFECTO - RECOMENDADO)
    ANGULO_CODO_ESTRICTO = (82, 98)

    # RELAJADO: 90° ± 20 grados (alternativa si necesitas más tolerancia)
    ANGULO_CODO_RELAJADO = (70, 110)

    # Rango activo (cambiar entre ESTRICTO y RELAJADO aquí)
    ANGULO_CODO_OBJ = ANGULO_CODO_ESTRICTO

PARA CAMBIAR DE NIVEL:

    # Opción 1: Cambiar globalmente
    import src.config as config
    config.ANGULO_CODO_OBJ = config.ANGULO_CODO_RELAJADO

    # Opción 2: Llamada específica
    from src.geometry import es_postura_L
    es_postura_L(h, c, m, w, nivel="relajado")

    # Opción 3: Modificar rangos
    config.ANGULO_CODO_ESTRICTO = (85, 95)  # ±5° muy estricto
